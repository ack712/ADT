CREATE DATABASE SaptialDB;
USE SaptialDB;

CREATE TABLE TouristPlaces(
PlaceID INT PRIMARY KEY,
PlaceName VARCHAR(50),
Location POINT NOT NULL
);

*Tourist Routes 
CREATE DATABASE SaptialDB1;
USE SaptialDB1;

CREATE TABLE TouristRoutes (
    route_id INT PRIMARY KEY,
    route_name VARCHAR(100),
    route LINESTRING NOT NULL
);
INSERT INTO TouristRoutes(route_id, route_name, route)
VALUES
(
    1,
    'Ratnagiri Tourist Route',
    ST_GeomFromText(
        'LINESTRING(
        73.3060 16.9944,
        73.3050 16.9990,
        73.2660 17.1448,
        73.2270 17.2930
        )'
    )
);
SELECT
    route_name,
    ST_AsText(route) AS Tourist_Route
FROM TouristRoutes;

SELECT
    route_name,
    ST_Length(route) AS Route_Length
FROM TouristRoutes;

SELECT
    route_name,
    ST_NumPoints(route) AS Total_Tourist_Points
FROM TouristRoutes;

SELECT
    route_name,
    ST_AsText(ST_StartPoint(route)) AS Start_Point
FROM TouristRoutes;

SELECT
    route_name,
    ST_AsText(ST_EndPoint(route)) AS End_Point
FROM TouristRoutes;

SELECT
    ST_AsText(ST_PointN(route, 2)) AS Second_Tourist_Point
FROM TouristRoutes;
