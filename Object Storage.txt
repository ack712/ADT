CREATE DATABASE ObjectDB;
USE ObjectDB;

CREATE TABLE Objects (
    id INT AUTO_INCREMENT PRIMARY KEY,
    file_name VARCHAR(100),
    file_type VARCHAR(50),
    file_data LONGBLOB
);

INSERT INTO Objects (file_name, file_type, file_data)
VALUES (
    'photo.jpg',
    'image/jpeg',
    LOAD_FILE('C:/mysql_files/photo.jpg')
);

SELECT id, file_name, file_type,
       LENGTH(file_data) AS file_size_bytes
FROM Objects;

SELECT id, file_name, file_type
FROM Objects;

SELECT file_name,
       LENGTH(file_data) AS size_bytes
FROM Objects;

UPDATE Objects
SET file_data = LOAD_FILE('C:/mysql_files/newphoto.jpg'),
    file_name = 'newphoto.jpg',
    file_type = 'image/jpeg'
WHERE id = 1;

SELECT COUNT(*) AS total_objects
FROM Objects;
