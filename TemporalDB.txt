CREATE DATABASE TemporalDB;
USE TemporalDB;

CREATE TABLE Customer_Temporal
(
    Customer_ID INT,
    Customer_Name VARCHAR(100),
    Address VARCHAR(100),
    Phone VARCHAR(15),
    Valid_From DATE,
    Valid_To DATE,
    Transaction_Start TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    Transaction_End TIMESTAMP NULL,
    PRIMARY KEY(Customer_ID, Valid_From)
);
